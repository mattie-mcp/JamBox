# JamBox Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Build JamBox — a smart NFC-based music player where Jamie places a toy on a pedestal and the corresponding Spotify track plays through a Bluetooth speaker.

**Architecture:** A Raspberry Pi runs three services: (1) an NFC listener daemon that detects tags and triggers Spotify playback via the Spotify Web API, (2) a Raspotify instance that makes the Pi a Spotify Connect device, and (3) a web app for parents to manage NFC tag-to-song mappings stored in SQLite. All services run on the Pi, accessed over the local network.

**Tech Stack:** Python 3 (NFC daemon + API backend), Next.js (web app), SQLite (data store), Spotify Web API (playback control), Raspotify (Spotify Connect), nfcpy (NFC reader library)

```markdown
## Wiring: PN532 to Raspberry Pi (I2C mode)

| PN532 Pin | RPi Pin | RPi GPIO |
|-----------|---------|----------|
| VCC       | Pin 1   | 3.3V     |
| GND       | Pin 6   | Ground   |
| SDA       | Pin 3   | GPIO 2 (SDA) |
| SCL       | Pin 5   | GPIO 3 (SCL) |
```

```python
import spotipy


class SpotifyService:
    def __init__(self, sp: spotipy.Spotify, device_name: str = "raspotify"):
        self.sp = sp
        self.device_name = device_name

    def _get_device_id(self) -> str:
        devices = self.sp.devices()
        for device in devices.get("devices", []):
            if device["name"] == self.device_name:
                return device["id"]
        raise RuntimeError(
            f"Spotify device '{self.device_name}' not found. "
            f"Available: {[d['name'] for d in devices.get('devices', [])]}"
        )

    def play_track(self, spotify_uri: str) -> None:
        device_id = self._get_device_id()
        self.sp.start_playback(device_id=device_id, uris=[spotify_uri])

    def pause(self) -> None:
        device_id = self._get_device_id()
        self.sp.pause_playback(device_id=device_id)

    def search_tracks(self, query: str, limit: int = 10) -> list[dict]:
        results = self.sp.search(q=query, type="track", limit=limit)
        tracks = []
        for item in results["tracks"]["items"]:
            tracks.append(
                {
                    "uri": item["uri"],
                    "name": item["name"],
                    "artist": ", ".join(a["name"] for a in item["artists"]),
                    "image_url": (
                        item["album"]["images"][0]["url"]
                        if item["album"]["images"]
                        else None
                    ),
                }
            )
        return tracks
```

```python
import os

import spotipy
from spotipy.oauth2 import SpotifyOAuth
from fastapi import APIRouter, Request
from fastapi.responses import RedirectResponse, JSONResponse

from app.config import SPOTIFY_CLIENT_ID, SPOTIFY_CLIENT_SECRET, SPOTIFY_REDIRECT_URI

router = APIRouter(prefix="/auth")

SCOPE = "user-read-playback-state user-modify-playback-state"

_token_cache_path = os.path.join(
    os.path.dirname(os.path.dirname(__file__)), ".spotify_cache"
)


def get_auth_manager() -> SpotifyOAuth:
    return SpotifyOAuth(
        client_id=SPOTIFY_CLIENT_ID,
        client_secret=SPOTIFY_CLIENT_SECRET,
        redirect_uri=SPOTIFY_REDIRECT_URI,
        scope=SCOPE,
        cache_path=_token_cache_path,
    )


def get_spotify() -> spotipy.Spotify:
    auth_manager = get_auth_manager()
    token_info = auth_manager.get_cached_token()
    if not token_info:
        raise RuntimeError("Not authenticated. Visit /auth/login first.")
    return spotipy.Spotify(auth_manager=auth_manager)


@router.get("/login")
def login():
    auth_manager = get_auth_manager()
    auth_url = auth_manager.get_authorize_url()
    return RedirectResponse(auth_url)


@router.get("/callback")
def callback(request: Request):
    code = request.query_params.get("code")
    if not code:
        return JSONResponse({"error": "Missing authorization code"}, status_code=400)
    auth_manager = get_auth_manager()
    auth_manager.get_access_token(code)
    return RedirectResponse("/")


@router.get("/status")
def auth_status():
    auth_manager = get_auth_manager()
    token_info = auth_manager.get_cached_token()
    return {"authenticated": token_info is not None}
```

```python
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.database import get_db
from app.models import TagMapping
from app.schemas import MappingCreate, MappingResponse, MappingUpdate

router = APIRouter(prefix="/api")


@router.get("/mappings", response_model=list[MappingResponse])
def list_mappings(db: Session = Depends(get_db)):
    return db.query(TagMapping).all()


@router.post("/mappings", response_model=MappingResponse, status_code=201)
def create_mapping(body: MappingCreate, db: Session = Depends(get_db)):
    mapping = TagMapping(
        tag_id=body.tag_id, spotify_uri=body.spotify_uri, name=body.name
    )
    db.add(mapping)
    try:
        db.commit()
    except IntegrityError:
        db.rollback()
        raise HTTPException(409, "A mapping for this tag already exists")
    db.refresh(mapping)
    return mapping


@router.put("/mappings/{mapping_id}", response_model=MappingResponse)
def update_mapping(
    mapping_id: int, body: MappingUpdate, db: Session = Depends(get_db)
):
    mapping = db.query(TagMapping).filter(TagMapping.id == mapping_id).first()
    if not mapping:
        raise HTTPException(404, "Mapping not found")
    if body.spotify_uri is not None:
        mapping.spotify_uri = body.spotify_uri
    if body.name is not None:
        mapping.name = body.name
    db.commit()
    db.refresh(mapping)
    return mapping


@router.delete("/mappings/{mapping_id}", status_code=204)
def delete_mapping(mapping_id: int, db: Session = Depends(get_db)):
    mapping = db.query(TagMapping).filter(TagMapping.id == mapping_id).first()
    if not mapping:
        raise HTTPException(404, "Mapping not found")
    db.delete(mapping)
    db.commit()
```

Note: This cannot be fully tested without hardware. We write it with clear separation so the core logic is testable.

```python
"""
NFC Listener Daemon

Polls the NFC reader and triggers playback via the API when a tag is detected.
Run on the Raspberry Pi: python -m app.nfc_daemon
"""

import time
import logging
import httpx

logging.basicConfig(level=logging.INFO, format="%(asctime)s %(message)s")
logger = logging.getLogger(__name__)

API_BASE = "http://localhost:8000"
POLL_INTERVAL = 0.5  # seconds


def trigger_play(tag_id: str) -> None:
    """Call the local API to play the track mapped to this tag."""
    try:
        response = httpx.post(f"{API_BASE}/api/play/{tag_id}", timeout=5)
        if response.status_code == 200:
            data = response.json()
            logger.info(f"Playing: {data['name']}")
        elif response.status_code == 404:
            logger.warning(f"No mapping for tag: {tag_id}")
        else:
            logger.error(f"API error: {response.status_code}")
    except httpx.RequestError as e:
        logger.error(f"Could not reach API: {e}")


def format_tag_id(tag_bytes: bytes) -> str:
    """Convert raw tag bytes to colon-separated hex string."""
    return ":".join(f"{b:02X}" for b in tag_bytes)


def run():
    """Main loop - requires nfcpy and hardware."""
    try:
        import nfc
    except ImportError:
        logger.error("nfcpy not installed. Run: pip install nfcpy")
        return

    last_tag = None
    last_tag_time = 0
    debounce_seconds = 5  # Don't re-trigger same tag within 5s

    def on_connect(tag):
        nonlocal last_tag, last_tag_time
        tag_id = format_tag_id(tag.identifier)
        now = time.time()

        if tag_id == last_tag and (now - last_tag_time) < debounce_seconds:
            return True  # Keep listening, ignore duplicate

        logger.info(f"Tag detected: {tag_id}")
        last_tag = tag_id
        last_tag_time = now
        trigger_play(tag_id)
        return True  # Keep listening

    with nfc.ContactlessFrontend("tty:S0") as clf:
        logger.info("NFC daemon started. Waiting for tags...")
        while True:
            clf.connect(rdwr={"on-connect": on_connect})
            time.sleep(POLL_INTERVAL)


if __name__ == "__main__":
    run()
```

```typescript
const API_BASE = process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000";

export interface TagMapping {
  id: number;
  tag_id: string;
  spotify_uri: string;
  name: string;
  created_at: string;
}

export interface SpotifyTrack {
  uri: string;
  name: string;
  artist: string;
  image_url: string | null;
}

export async function fetchMappings(): Promise<TagMapping[]> {
  const res = await fetch(`${API_BASE}/api/mappings`);
  if (!res.ok) throw new Error("Failed to fetch mappings");
  return res.json();
}

export async function createMapping(data: {
  tag_id: string;
  spotify_uri: string;
  name: string;
}): Promise<TagMapping> {
  const res = await fetch(`${API_BASE}/api/mappings`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(data),
  });
  if (!res.ok) throw new Error("Failed to create mapping");
  return res.json();
}

export async function deleteMapping(id: number): Promise<void> {
  const res = await fetch(`${API_BASE}/api/mappings/${id}`, {
    method: "DELETE",
  });
  if (!res.ok) throw new Error("Failed to delete mapping");
}

export async function searchTracks(query: string): Promise<SpotifyTrack[]> {
  const res = await fetch(
    `${API_BASE}/api/search?q=${encodeURIComponent(query)}`
  );
  if (!res.ok) throw new Error("Failed to search");
  return res.json();
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import {
  fetchMappings,
  createMapping,
  deleteMapping,
  searchTracks,
  TagMapping,
  SpotifyTrack,
} from "@/lib/api";

export default function Home() {
  const [mappings, setMappings] = useState<TagMapping[]>([]);
  const [searchQuery, setSearchQuery] = useState("");
  const [searchResults, setSearchResults] = useState<SpotifyTrack[]>([]);
  const [selectedTrack, setSelectedTrack] = useState<SpotifyTrack | null>(null);
  const [tagId, setTagId] = useState("");
  const [error, setError] = useState("");
  const [loading, setLoading] = useState(true);

  async function loadMappings() {
    try {
      const data = await fetchMappings();
      setMappings(data);
    } catch {
      setError("Could not load mappings. Is the backend running?");
    } finally {
      setLoading(false);
    }
  }

  useEffect(() => {
    loadMappings();
  }, []);

  async function handleSearch() {
    if (!searchQuery.trim()) return;
    try {
      const results = await searchTracks(searchQuery);
      setSearchResults(results);
    } catch {
      setError("Search failed. Check Spotify authentication.");
    }
  }

  async function handleCreate() {
    if (!tagId.trim() || !selectedTrack) return;
    try {
      await createMapping({
        tag_id: tagId.trim(),
        spotify_uri: selectedTrack.uri,
        name: `${selectedTrack.name} - ${selectedTrack.artist}`,
      });
      setTagId("");
      setSelectedTrack(null);
      setSearchResults([]);
      setSearchQuery("");
      loadMappings();
    } catch {
      setError("Failed to create mapping. Tag may already be assigned.");
    }
  }

  async function handleDelete(id: number) {
    try {
      await deleteMapping(id);
      loadMappings();
    } catch {
      setError("Failed to delete mapping.");
    }
  }

  return (
    <main className="min-h-screen bg-gray-50 p-6 max-w-2xl mx-auto">
      <h1 className="text-3xl font-bold mb-6">JamBox</h1>

      {error && (
        <div className="bg-red-100 border border-red-300 text-red-700 rounded p-3 mb-4">
          {error}
          <button className="ml-2 underline" onClick={() => setError("")}>
            dismiss
          </button>
        </div>
      )}

      {/* Current Mappings */}
      <section className="mb-8">
        <h2 className="text-xl font-semibold mb-3">Tag Mappings</h2>
        {loading ? (
          <p className="text-gray-500">Loading...</p>
        ) : mappings.length === 0 ? (
          <p className="text-gray-500">No mappings yet. Add one below.</p>
        ) : (
          <ul className="space-y-2">
            {mappings.map((m) => (
              <li
                key={m.id}
                className="flex items-center justify-between bg-white rounded-lg shadow p-4"
              >
                <div>
                  <p className="font-medium">{m.name}</p>
                  <p className="text-sm text-gray-500">Tag: {m.tag_id}</p>
                </div>
                <button
                  onClick={() => handleDelete(m.id)}
                  className="text-red-500 hover:text-red-700 text-sm"
                >
                  Remove
                </button>
              </li>
            ))}
          </ul>
        )}
      </section>

      {/* Add New Mapping */}
      <section className="bg-white rounded-lg shadow p-6">
        <h2 className="text-xl font-semibold mb-4">Add New Mapping</h2>

        {/* Tag ID Input */}
        <div className="mb-4">
          <label className="block text-sm font-medium mb-1">NFC Tag ID</label>
          <input
            type="text"
            value={tagId}
            onChange={(e) => setTagId(e.target.value)}
            placeholder="e.g. 04:A2:3B:5C:6D:7E:80"
            className="w-full border rounded px-3 py-2"
          />
        </div>

        {/* Spotify Search */}
        <div className="mb-4">
          <label className="block text-sm font-medium mb-1">
            Search Spotify
          </label>
          <div className="flex gap-2">
            <input
              type="text"
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              onKeyDown={(e) => e.key === "Enter" && handleSearch()}
              placeholder="Search for a song..."
              className="flex-1 border rounded px-3 py-2"
            />
            <button
              onClick={handleSearch}
              className="bg-green-600 text-white px-4 py-2 rounded hover:bg-green-700"
            >
              Search
            </button>
          </div>
        </div>

        {/* Search Results */}
        {searchResults.length > 0 && (
          <div className="mb-4 max-h-60 overflow-y-auto">
            <ul className="space-y-1">
              {searchResults.map((track) => (
                <li
                  key={track.uri}
                  onClick={() => setSelectedTrack(track)}
                  className={`flex items-center gap-3 p-2 rounded cursor-pointer hover:bg-gray-100 ${
                    selectedTrack?.uri === track.uri
                      ? "bg-green-50 border border-green-300"
                      : ""
                  }`}
                >
                  {track.image_url && (
                    <img
                      src={track.image_url}
                      alt=""
                      className="w-10 h-10 rounded"
                    />
                  )}
                  <div>
                    <p className="text-sm font-medium">{track.name}</p>
                    <p className="text-xs text-gray-500">{track.artist}</p>
                  </div>
                </li>
              ))}
            </ul>
          </div>
        )}

        {/* Selected Track */}
        {selectedTrack && (
          <div className="mb-4 p-3 bg-green-50 rounded">
            Selected: <strong>{selectedTrack.name}</strong> by{" "}
            {selectedTrack.artist}
          </div>
        )}

        <button
          onClick={handleCreate}
          disabled={!tagId.trim() || !selectedTrack}
          className="w-full bg-blue-600 text-white py-2 rounded hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
        >
          Save Mapping
        </button>
      </section>
    </main>
  );
}
```
